# ===========================================================================
#  SUI example — "Flow Field"
#  A generative particle simulation drawn each frame with script(). Thousands
#  of particles drift along a smooth noise field, leaving fading trails that
#  recolour as they move. The sidebar controls the simulation.
#
#  Run:  python box.py examples/flow.txt
# ===========================================================================

# ------ colors ------
BG =            (9, 15, 17, 255)
PANEL =         (18, 28, 30, 255)

# ------ simulation logic + helpers ------
@python
import math, random, time
from math import sin, cos, pi

FLOW = {
    "n": 260, "speed": 1.6, "scale": 0.004, "sat": 0.78, "val": 0.92,
    "alive": True, "hue": 0.0, "t": 0.0, "paths": None, "pathlen": 20,
}

def flow_reset(box):
    w = max(2.0, box.rect.width)
    h = max(2.0, box.rect.height)
    FLOW["paths"] = [[[random.uniform(0, w), random.uniform(0, h)]] for _ in range(FLOW["n"])]

def flow_frame(box):
    w = box.rect.width
    h = box.rect.height
    if w <= 2 or h <= 2:
        return
    x0 = box.rect.x
    y0 = box.rect.y
    if FLOW["paths"] is None or len(FLOW["paths"]) != FLOW["n"]:
        flow_reset(box)
    draw_rectangle(int(x0), int(y0), int(w), int(h), Color(9, 15, 17, 255))
    if FLOW["alive"]:
        FLOW["t"] += 0.02
    t = FLOW["t"]
    L = FLOW["pathlen"]
    sc = FLOW["scale"]
    spd = FLOW["speed"]
    hue0 = FLOW["hue"]
    for p in FLOW["paths"]:
        px, py = p[-1]
        ang = (sin(px * sc + t * 0.5) + sin(py * sc * 1.4 - t * 0.35)
               + sin((px + py) * sc * 0.6 + t * 0.15)) * pi
        ix = px + cos(ang) * spd
        iy = py + sin(ang) * spd
        nx = ix % w
        ny = iy % h
        if ix < 0 or ix >= w or iy < 0 or iy >= h:
            # particle wrapped around the edge -> restart its trail so the
            # trail never draws a long line all the way across the screen
            p[:] = [[nx, ny]]
        else:
            p.append([nx, ny])
            if len(p) > L:
                p.pop(0)
        m = len(p)
        base = color_from_hsv((hue0 + px * 0.05 + py * 0.02) % 360, FLOW["sat"], FLOW["val"])
        # draw every 2nd point (fading toward the tail) for speed
        for i in range(2, m, 2):
            c = fade(base, i / m)
            draw_line_ex(Vector2(x0 + p[i - 2][0], y0 + p[i - 2][1]),
                         Vector2(x0 + p[i][0], y0 + p[i][1]), 1.3, c)
    if FLOW["alive"]:
        FLOW["hue"] = (FLOW["hue"] + 0.25) % 360

def flow_toggle(box):
    FLOW["alive"] = not FLOW["alive"]
    return True

def flow_clear(box):
    FLOW["paths"] = None
    FLOW["hue"] = 0.0
    FLOW["t"] = 0.0
    return True

def flow_n(box, d):
    FLOW["n"] = max(40, min(4000, int(FLOW["n"] + d)))
    FLOW["paths"] = None
    return True

def flow_speed(box, d):
    FLOW["speed"] = max(0.2, min(9.0, FLOW["speed"] + float(d)))
    return True

def flow_preset(box, idx):
    if idx == 0:
        FLOW.update({"scale": 0.004, "speed": 1.4, "sat": 0.78, "val": 0.92, "pathlen": 24})
    elif idx == 1:
        FLOW.update({"scale": 0.009, "speed": 3.6, "sat": 0.97, "val": 0.97, "pathlen": 34})
    else:
        FLOW.update({"scale": 0.0014, "speed": 0.7, "sat": 0.55, "val": 0.85, "pathlen": 16})
    FLOW["paths"] = None
    return True

def flow_status():
    return "particles %d  speed %.1f  %s" % (
        FLOW["n"], FLOW["speed"], "flowing" if FLOW["alive"] else "paused")
@end
