# ===========================================================================
#  SUI example — "Glassurf" / "Lens Control"
#  The Waves shader is a live wallpaper; on top, a row of lens buttons each
#  refRACT the animation behind them (a convex bulge, prismatic rim, specular
#  sheen that tracks the mouse). Clicking a lens bumps the counter via a python
#  callback.
#
#  Run:  python box.py examples/glassurf.txt
# ===========================================================================

@python
import datetime

def clk():
    return datetime.datetime.now().strftime("%H:%M:%S")

STATE = {"n": 0, "frost": 0.0, "opacity": 0.55}

def clampv(v, lo, hi):
    return max(lo, min(hi, v))

def bump(box, d):
    STATE["n"] = int(clampv(STATE["n"] + int(d), 0, 60))
    return True

def count():
    return "refraction  %d" % STATE["n"]

def refn():
    return STATE["n"]

def set_frost(box, d):
    STATE["frost"] = round(clampv(STATE["frost"] + float(d), 0.0, 1.0), 2)
    return True

def frost_val():
    return STATE["frost"]

def frost_label():
    return "frost %d%%" % round(STATE["frost"] * 100)

def set_opacity(box, d):
    STATE["opacity"] = round(clampv(STATE["opacity"] + float(d), 0.1, 0.95), 2)
    return True

def opacity_val():
    return STATE["opacity"]

def opacity_label():
    return "opacity %d%%" % round(STATE["opacity"] * 100)
@end
